static bool getFreshEnvSnapshot(EnvResponse& out, uint32_t maxAgeMs) {
    if (!g_envValid) return false;
    const uint32_t ageMs = (uint32_t)(millis() - g_envRecvMs);
    if (ageMs > maxAgeMs) return false;

    // Copy with a seq-stability check (ESP-NOW callback runs in another task).
    uint16_t seq1 = g_lastEnv.seq;
    out = g_lastEnv;
    uint16_t seq2 = g_lastEnv.seq;
    if (seq1 != seq2) {
        out = g_lastEnv;
    }
    return out.type == MSG_ENV_RESPONSE;
}

    while (millis() - start < timeoutMs) {
        feed_wdt();
        if (g_envValid) return true;
        delay(5);
    }
    return false;
}

// HTTPS operations can block long enough to trip the task watchdog.
// We temporarily remove the current task from TWDT during those calls.
struct WdtSuspendGuard {
    bool suspended = false;

    WdtSuspendGuard() {
        if (!g_wdt_task_added) return;
        esp_err_t err = esp_task_wdt_delete(NULL);
        if (err == ESP_OK) {
            g_wdt_task_added = false;
            suspended = true;
        }
    }

    ~WdtSuspendGuard() {
        if (!suspended) return;
        esp_err_t err = esp_task_wdt_add(NULL);
        if (err == ESP_OK) {
            g_wdt_task_added = true;
        }
    }
};

static inline void feed_wdt() {
    if (g_wdt_task_added) {
        esp_task_wdt_reset();
    }
}

static bool safe_snprintf_append(char* dst, size_t dst_size, size_t& offset, const char* fmt, ...) {
    if (!dst || dst_size == 0 || offset >= dst_size) return false;
    va_list args;
    va_start(args, fmt);
    int written = vsnprintf(dst + offset, dst_size - offset, fmt, args);
    va_end(args);
    if (written < 0) return false;
    if ((size_t)written >= (dst_size - offset)) return false;
    offset += (size_t)written;
    return true;
}

static bool supabase_extract_missing_column(const String& body, String& outColumn) {
    // Note: PGRST message format is: Could not find the 'col' column ...
    int start = body.indexOf("Could not find the '");
    if (start < 0) return false;
    start += (int)strlen("Could not find the '");
    int end = body.indexOf("' column", start);
    if (end < 0) {
        end = body.indexOf("'", start);
        if (end < 0) return false;
    }
    outColumn = body.substring(start, end);
    outColumn.trim();
    return outColumn.length() > 0;
}

static int supabase_post_with_schema_fallback(HTTPClient& http, DynamicJsonDocument& doc, const char* tag) {
    for (int attempt = 1; attempt <= 4; attempt++) {
        String payload;
        payload.reserve(1024);
        serializeJson(doc, payload);

        feed_wdt();
        int code = http.POST(payload);
        if (code == 200 || code == 201) {
            return code;
        }

        String body;
        if (code > 0) {
            body = http.getString();
        }

        if (code == 400 && body.indexOf("\"code\":\"PGRST204\"") >= 0) {
            String missing;
            if (supabase_extract_missing_column(body, missing)) {
                Serial.printf("[%s] Missing column '%s' in table '%s' — dropping and retrying (%d/4)\n",
                              tag,
                              missing.c_str(),
                              SUPABASE_SENSOR_DATA_TABLE,
                              attempt);
                JsonObject root = doc.as<JsonObject>();
                root.remove(missing.c_str());
                continue;
            }
        }

        Serial.printf("[%s] Supabase POST failed (HTTP %d)\n", tag, code);
        if (code > 0 && body.length() > 0) {
            Serial.println(body);
        }
        return code;
    }
    return -1;
}

static void updateSpO2() {
    uint32_t irBuffer[30];
    uint32_t redBuffer[30];

    for (byte i = 0; i < 30; i++) {
        unsigned long startWait = millis();
        while (!maxSensor.available()) {
            maxSensor.check();
            if (millis() - startWait > 100) break;
        }
        irBuffer[i] = maxSensor.getIR();
        redBuffer[i] = maxSensor.getRed();
        maxSensor.nextSample();
    }

    uint32_t irMax = 0, irMin = 0xFFFFFFFF;
    uint32_t redMax = 0, redMin = 0xFFFFFFFF;

    for (byte i = 0; i < 30; i++) {
        if (irBuffer[i] > irMax) irMax = irBuffer[i];
        if (irBuffer[i] < irMin) irMin = irBuffer[i];
        if (redBuffer[i] > redMax) redMax = redBuffer[i];
        if (redBuffer[i] < redMin) redMin = redBuffer[i];
    }

    float f_irAC = (float)(irMax - irMin);
    float f_redAC = (float)(redMax - redMin);

    if (irMin > 0 && f_irAC > 0 && redMin > 0) {
        float ratio = (f_redAC / (float)redMin) / (f_irAC / (float)irMin);
        float raw_spo2 = -45.060f * ratio * ratio + 30.354f * ratio + 94.845f;

        // NEW: Smoothing and Validation Logic
        if (raw_spo2 > 100) raw_spo2 = 100;

        // Only accept the reading if it's realistic (85-100) to filter out noise
        if (raw_spo2 > 70.0f) {
            spo2Buffer[spo2Spot++] = raw_spo2;
            spo2Spot %= SPO2_SIZE;

            float spo2Sum = 0;
            for (byte x = 0; x < SPO2_SIZE; x++) {
                // Initialize buffer with first valid reading if empty
                if (spo2Buffer[x] == 0) spo2Buffer[x] = raw_spo2;
                spo2Sum += spo2Buffer[x];
            }
            current_spo2 = spo2Sum / SPO2_SIZE;
        } else {
            // If the reading is suspiciously low (like 73%), we ignore it
            // and keep the last known good current_spo2.
            Serial.println("[SpO2] Discarded noisy reading.");
        }
    }
}

static String captureAndClassifyAudio() {
    if (!audio_buffer) return "normal";
    Serial.println("[Audio] Capturing 3-second sample...");

    size_t bytes_read = 0;
    int32_t sample_buffer[256];
    int16_t* ptr = audio_buffer;
    int samples_collected = 0;

    while (samples_collected < SAMPLES) {
        int to_read = (SAMPLES - samples_collected > 256) ? 256 : (SAMPLES - samples_collected);
        i2s_read(I2S_PORT, sample_buffer, (size_t)to_read * 4, &bytes_read, portMAX_DELAY);
        int read_count = (int)(bytes_read / 4);
        for (int j = 0; j < read_count; j++) {
            *ptr++ = (int16_t)(sample_buffer[j] >> 14);
        }
        samples_collected += read_count;
        feed_wdt();
    }

    float rms = 0.0f;
    for (int i = 0; i < SAMPLES; i++) {
        rms += (float)audio_buffer[i] * (float)audio_buffer[i];
    }
    rms = sqrtf(rms / (float)SAMPLES);
    Serial.printf("[Audio] RMS Energy: %.2f\n", rms);

    if (rms < 100.0f || WiFi.status() != WL_CONNECTED) {
        Serial.println("[Audio] Signal too quiet or WiFi lost.");
        return "normal";
    }

    const size_t headerReserve = 64;
    const size_t tailReserve = 4;
    const size_t worstCharsPerSample = 7;
    size_t json_size = headerReserve + ((size_t)SAMPLES * worstCharsPerSample) + tailReserve;
    json_size = (json_size + 1023) & ~((size_t)1023);

    char* json_payload = (char*)heap_caps_malloc(json_size, MALLOC_CAP_SPIRAM);
    if (!json_payload) return "normal";

    size_t offset = 0;
    if (!safe_snprintf_append(json_payload, json_size, offset, "{\"device_id\":\"hikon_s3_v1\",\"data\":[")) {
        heap_caps_free(json_payload);
        return "normal";
    }

    float gain = 2.0f;
    for (int i = 0; i < SAMPLES; i++) {
        int32_t amplified = (int32_t)((float)audio_buffer[i] * gain);
        if (amplified > 32767) amplified = 32767;
        if (amplified < -32768) amplified = -32768;

        if (!safe_snprintf_append(json_payload, json_size, offset, "%d", (int16_t)amplified)) {
            Serial.println("[Audio] JSON overflow - skipping upload");
            heap_caps_free(json_payload);
            return "normal";
        }

        if (i < SAMPLES - 1) {
            if (offset + 1 >= json_size) {
                Serial.println("[Audio] JSON overflow (comma) - skipping upload");
                heap_caps_free(json_payload);
                return "normal";
            }
            json_payload[offset++] = ',';
            json_payload[offset] = '\0';
        }
    }
    if (!safe_snprintf_append(json_payload, json_size, offset, "]}")) {
        heap_caps_free(json_payload);
        return "normal";
    }

    const int MAX_RETRIES = 3;
    const unsigned long RETRY_DELAY_MS = 30000UL;

    int code = -1;
    String responseBody;
    String pred = "normal";

    for (int attempt = 1; attempt <= MAX_RETRIES; attempt++) {
        if (WiFi.status() != WL_CONNECTED) {
            Serial.println("[Audio] WiFi lost — reconnecting...");
            WiFi.reconnect();
            unsigned long start = millis();
            while (WiFi.status() != WL_CONNECTED && millis() - start < 10000UL) {
                delay(500);
                feed_wdt();
            }
        }

        WiFiClientSecure audioClient;
        HTTPClient audioHttp;
        audioClient.setInsecure();
        audioClient.setTimeout(45000);

        Serial.printf("[Audio] Sending to cloud (attempt %d/%d)...\n", attempt, MAX_RETRIES);
        audioHttp.begin(audioClient, VERCEL_API_URL);
        audioHttp.addHeader("Content-Type", "application/json");
        audioHttp.setConnectTimeout(15000);
        audioHttp.setTimeout(45000);
        feed_wdt();

        // HTTPS POST can block for TLS handshake / cold start.
        // Suspend TWDT for the current task so it doesn't reboot mid-request.
        {
            WdtSuspendGuard wdtGuard;
            code = audioHttp.POST((uint8_t*)json_payload, offset);
        }

        if (code == 200 || code == 201) {
            responseBody = audioHttp.getString();
            audioHttp.end();
            audioClient.stop();
            break;
        }

        Serial.printf("[Audio] Cloud inference failed (HTTP %d)\n", code);
        if (code > 0) {
            responseBody = audioHttp.getString();
            Serial.println("[Audio] Response: " + responseBody);
        }

        audioHttp.end();
        audioClient.stop();

        bool isConfigError = (code >= 400 && code < 500);
        if (isConfigError) {
            Serial.println("[Audio] Config error — not retrying");
            break;
        }

        if (attempt < MAX_RETRIES) {
            Serial.printf("[Audio] Cloud may be waking up — waiting %lu sec before retry...\n", RETRY_DELAY_MS / 1000UL);
            unsigned long waitUntil = millis() + RETRY_DELAY_MS;
            while (millis() < waitUntil) {
                delay(100);
                feed_wdt();
            }
        }
    }

    heap_caps_free(json_payload);

    if (code == 200 || code == 201) {
        Serial.println("[Audio] Full response: " + responseBody);

        DynamicJsonDocument d(1024);
        DeserializationError err = deserializeJson(d, responseBody);
        if (err) {
            Serial.printf("[Audio] JSON parse error: %s\n", err.c_str());
            return "normal";
        }

        int cough = d["inference"]["cough"] | 0;
        int wheeze = d["inference"]["wheeze"] | 0;
        bool gemini_verified = d["gemini_verified"] | false;
        const char* gemini_error = d["gemini_error"] | "";

        if (cough == 1) pred = "cough";
        else if (wheeze == 1) pred = "wheeze";

        Serial.printf("[Audio] Result: %s (Check: %s)\n", pred.c_str(), gemini_verified ? "GEMINI VERIFIED" : "HF-ONLY");
        if (gemini_error && strlen(gemini_error) > 0) {
            Serial.printf("[Audio] Gemini Error: %s\n", gemini_error);
        }
        return pred;
    }

    Serial.println("[Audio] Inference failed — defaulting to normal");
    return "normal";
}

static String calculatePhysioRisk(float hr, float spo2, const String& audioLabel) {
    // Physiological Assessment (Triage System)
    bool hrValid = (hr > 40.0f && hr < 180.0f);
    bool spo2Valid = (spo2 > 70.0f && spo2 <= 100.0f);

    if (!hrValid || !spo2Valid) {
        return "error";
    }

    if (hr > 140.0f || spo2 < 88.0f || audioLabel == "cough" || audioLabel == "wheeze") {
        return "critical";
    }
    if (hr > 115.0f) {
        return "medium";
    }
    return "safe";
}

static void printMac6(const uint8_t* mac) {
    if (!mac) {
        Serial.print("(null)");
        return;
    }
    Serial.printf("%02X:%02X:%02X:%02X:%02X:%02X", mac[0], mac[1], mac[2], mac[3], mac[4], mac[5]);
}

static void onDataSent(const uint8_t* mac_addr, esp_now_send_status_t status) {
    Serial.print("[ESP-NOW] Send status to ");
    printMac6(mac_addr);
    Serial.print(": ");
    Serial.println(status == ESP_NOW_SEND_SUCCESS ? "SUCCESS" : "FAIL");
}

// ESP-NOW recv callback signature differs between Arduino-ESP32 core versions.
// - core v2.x (IDF4): (mac, data, len)
// - core v3.x (IDF5): (recv_info, data, len)
#if ESP_IDF_VERSION_MAJOR >= 5
void onDataRecv(const esp_now_recv_info_t* recv_info, const uint8_t* incomingData, int len) {
    const uint8_t* src = recv_info ? recv_info->src_addr : nullptr;
    if (len > 0 && incomingData && incomingData[0] == MSG_READY_ACK) {
        partnerReady = true;
        if (src) {
            Serial.print("[Handshake] READY_ACK received from ");
            printMac6(src);
            Serial.println(" — Board B is ONLINE. Starting system...");
        } else {
            Serial.println("[Handshake] Board B is ONLINE. Starting system...");
        }
        return;
    }

    if (incomingData && len == (int)sizeof(EnvResponse) && incomingData[0] == MSG_ENV_RESPONSE) {
        EnvResponse r;
        memcpy(&r, incomingData, sizeof(r));
        g_lastEnv = r;
        g_envRecvMs = millis();
        g_envValid = true;
        Serial.printf("[Env] Snapshot received: T=%.1f H=%.1f PM=%u/%u/%u\n",
                      ((float)r.temp_c_x10) / 10.0f,
                      ((float)r.hum_rh_x10) / 10.0f,
                      (unsigned)r.pm10, (unsigned)r.pm25, (unsigned)r.pm100);
    }
}
#else

    (void)mac;
    if (len > 0 && incomingData && incomingData[0] == MSG_READY_ACK) { // 0xAC is our "I'm ready" code

/* =============== Sleep / Wake Implementation =============== */
static void sendSleepCommand() {
    const uint8_t cmd = MSG_SLEEP_CMD;
    esp_now_send(boardB_Address, &cmd, sizeof(cmd));
    delay(150); // give Board B time to receive before we sleep
}

static void enterDeepSleep() {
    Serial.println("[Sleep] Entering deep sleep... press GP1 button to wake.");
    Serial.flush();

// Wait for button to be released before sleeping.
    // Otherwise the still-pressed button immediately wakes the chip.
    Serial.println("[Sleep] Release button to confirm sleep...");
    while (digitalRead(SLEEP_BUTTON_PIN) == LOW) {
        delay(10);
    }
    delay(200);

    // GP1 is an RTC GPIO — supports ext0 wakeup from deep sleep.
    // Level 0 = wake when pin goes LOW (button pressed, pulls GP1 to GND).
     esp_sleep_enable_ext0_wakeup((gpio_num_t)SLEEP_BUTTON_PIN, 0);
    esp_deep_sleep_start();
}

// Call this every loop iteration — handles long-press detection on GP1.
static void handleSleepButton() {
    bool pressed = (digitalRead(SLEEP_BUTTON_PIN) == LOW); // LOW = pressed

    if (pressed && !g_bootBtnHeld) {
        // Button just went down — start timing.
        g_bootBtnHeld      = true;
        g_bootBtnPressedMs = millis();
    } else if (!pressed && g_bootBtnHeld) {
        // Button released before long press threshold — ignore (short press).
        g_bootBtnHeld = false;
    } else if (pressed && g_bootBtnHeld) {
        // Button still held — check if long press threshold reached.
        if (millis() - g_bootBtnPressedMs >= LONG_PRESS_MS) {
            g_bootBtnHeld = false; // reset so it doesn't fire repeatedly
            Serial.println("[Sleep] Long press detected — sleeping...");
            sendSleepCommand();
            enterDeepSleep();
        }
    }
}

/* =============== Setup & Loop =============== */
void setup() {
    Serial.begin(115200);
    delay(1000);
    Serial.println("[HikOn] Booting...");

    // Watchdog (TWDT)
    esp_err_t wdt_err = ESP_OK;
#if ESP_IDF_VERSION_MAJOR >= 5
    const esp_task_wdt_config_t wdt_cfg = { .timeout_ms = 60000, .idle_core_mask = 0, .trigger_panic = false };
    wdt_err = esp_task_wdt_reconfigure(&wdt_cfg);
    if (wdt_err == ESP_ERR_INVALID_STATE) {
        // TWDT not initialized yet in this core build
        wdt_err = esp_task_wdt_init(&wdt_cfg);
    }
#else
    // Older cores only support init(timeout_seconds, panic)
    wdt_err = esp_task_wdt_init(60, false);
#endif
    if (wdt_err != ESP_OK && wdt_err != ESP_ERR_INVALID_STATE) {
        Serial.printf("[WDT] Config failed: %s\n", esp_err_to_name(wdt_err));
    }

    esp_err_t add_err = esp_task_wdt_add(NULL);
    if (add_err == ESP_OK) {
        g_wdt_task_added = true;
    } else {
        Serial.printf("[WDT] Add task failed: %s\n", esp_err_to_name(add_err));
    }

    // WiFi (ESP-NOW requires STA mode; set mode before begin)
    WiFi.mode(WIFI_STA);

    // Helps prevent missed ESP-NOW packets while associated to an AP.
    WiFi.setSleep(false);
#if __has_include(<esp_wifi.h>)
    (void)esp_wifi_set_ps(WIFI_PS_NONE);
#endif

    Serial.print("[Physio] MAC: ");
    Serial.println(WiFi.macAddress());
    WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
    Serial.print("[WiFi] Connecting");
    unsigned long wifiStart = millis();
    while (WiFi.status() != WL_CONNECTED && millis() - wifiStart < 20000UL) {
        delay(500);
        Serial.print('.');
        feed_wdt();
    }
    Serial.println();
    if (WiFi.status() == WL_CONNECTED) {
        Serial.printf("[WiFi] Connected (channel %u)\n", (unsigned)WiFi.channel());
    } else {
        Serial.println("[WiFi] Not connected (continuing)");
    }

#if __has_include(<esp_wifi.h>)
    // Re-apply no power-save and boost TX power after association.
    (void)esp_wifi_set_ps(WIFI_PS_NONE);
    esp_wifi_set_max_tx_power(84); // 21 dBm — compensates for XIAO Seeed ceramic antenna
#endif

    // Audio buffer (prefer PSRAM)
    audio_buffer = (int16_t*)heap_caps_malloc((size_t)SAMPLES * sizeof(int16_t), MALLOC_CAP_SPIRAM | MALLOC_CAP_8BIT);
    if (!audio_buffer) {
        audio_buffer = (int16_t*)malloc((size_t)SAMPLES * sizeof(int16_t));
    }
    if (!audio_buffer) {
        Serial.println("[Audio] Buffer allocation failed");
    }

    // I2C + sensors
    Wire.begin(SDA_PIN, SCL_PIN);
    max30102_connected = configureMAX30102Sensor();

    mpu.initialize();
    mpu6050_connected = mpu.testConnection();

    // Initialize ESP-NOW
    if (esp_now_init() != ESP_OK) {
        Serial.println("[ESP-NOW] Error initializing");
        return;
    }
    esp_now_register_recv_cb(onDataRecv);
    esp_now_register_send_cb(onDataSent);

    // Register Board B as a peer
    esp_now_peer_info_t peerInfo;
    memset(&peerInfo, 0, sizeof(peerInfo));
    memcpy(peerInfo.peer_addr, boardB_Address, 6);
    // Use channel=0 so the driver always uses the current STA channel automatically.
    peerInfo.channel = 0;
    peerInfo.encrypt = false;
#if ESP_IDF_VERSION_MAJOR >= 4
    peerInfo.ifidx = WIFI_IF_STA;
#endif
    if (esp_now_add_peer(&peerInfo) != ESP_OK) {
        Serial.println("[ESP-NOW] Failed to add peer");
    } else {
        Serial.printf("[ESP-NOW] Peer added on channel %u\n", (unsigned)peerInfo.channel);
    }

    // I2S mic
    i2s_config_t i2s_cfg;
    memset(&i2s_cfg, 0, sizeof(i2s_cfg));
    i2s_cfg.mode = (i2s_mode_t)(I2S_MODE_MASTER | I2S_MODE_RX);
    i2s_cfg.sample_rate = SAMPLE_RATE;
    i2s_cfg.bits_per_sample = I2S_BITS_PER_SAMPLE_32BIT;
    i2s_cfg.channel_format = I2S_CHANNEL_FMT_ONLY_LEFT;
    i2s_cfg.communication_format = I2S_COMM_FORMAT_STAND_I2S;
    i2s_cfg.dma_buf_count = 4;
    i2s_cfg.dma_buf_len = 512;
    i2s_cfg.use_apll = false;

    i2s_pin_config_t i2s_pins;
    memset(&i2s_pins, 0, sizeof(i2s_pins));
    i2s_pins.bck_io_num = I2S_SCK;
    i2s_pins.ws_io_num = I2S_WS;
    i2s_pins.data_out_num = I2S_PIN_NO_CHANGE;
    i2s_pins.data_in_num = I2S_SD;

    if (i2s_driver_install(I2S_PORT, &i2s_cfg, 0, NULL) == ESP_OK) {
        i2s_set_pin(I2S_PORT, &i2s_pins);
        Serial.println("[I2S] Ready");
    } else {
        Serial.println("[I2S] Init failed");
    }

    Serial.println("[System] Initialization complete.");

    // Configure GP1 as sleep button input with internal pull-up.
    // Button wired between GP1 and GND — pressed = LOW.
    pinMode(SLEEP_BUTTON_PIN, INPUT_PULLUP);
    Serial.println("[Sleep] Hold GP1 button 3 sec to sleep. Press to wake (reboots).");
}

void loop() {
    unsigned long now = millis();
    feed_wdt();

    // Check GP1 button for long-press sleep trigger.
    handleSleepButton();

    if (!partnerReady) {
        // Send a ping every second to find Board B
        if (now - lastPingTime >= PING_INTERVAL) {
            if (WiFi.status() != WL_CONNECTED) {
                Serial.println("[Handshake] Waiting for WiFi to stabilize before ESP-NOW...");
                WiFi.reconnect();
                lastPingTime = now;
                return;
            }
            uint8_t msg = MSG_PING_ARE_YOU_THERE; // "Are you there?" code
            esp_err_t err = esp_now_send(boardB_Address, &msg, sizeof(msg));
            if (err != ESP_OK) {
                Serial.printf("[Handshake] Ping send failed: %d\n", (int)err);
            }
            Serial.println("[Handshake] Searching for Board B...");
            lastPingTime = now;
        }
        return; // Skip the rest of the loop until Board B answers
    }

    // HR (real-time)
    if (max30102_connected) {
        long irValue = maxSensor.getIR();
        // Lowered threshold to 25k to catch more consistent peaks
        if (irValue > 25000 && checkForBeat(irValue)) {
            // First-beat guard: avoid calculating BPM from time 0
            if (lastBeat == 0) {
                lastBeat = millis();
                return;
            }

            long delta = millis() - lastBeat;
            lastBeat = millis();
            float rawBpm = 60.0f / (delta / 1000.0f);

            // Range guard: Discard impossible spikes (like the 300 BPM you saw)
            if (rawBpm > 40 && rawBpm < 185) {
                rates[rateSpot++] = (byte)rawBpm;
                rateSpot %= RATE_SIZE;

                // Calculate the average of the last 4 beats for a stable display
                float avg = 0;
                for (byte x = 0; x < RATE_SIZE; x++) avg += rates[x];
                current_hr = avg / RATE_SIZE;

                if (current_spo2 > 0) {
                    Serial.printf("[HR] %.1f | [SPO2] %.1f%% (Last Read) | IR: %ld\n", current_hr, current_spo2, irValue);
                } else {
                    Serial.printf("[HR] %.1f | [SPO2] Initializing... | IR: %ld\n", current_hr, irValue);
                }
            }
        }
    }

    // Periodic accel upload
if (now - lastAccelUpload >= ACCEL_UPLOAD_INTERVAL_MS) {
    uploadAccelToSupabase();
    // FIX: guard with max30102_connected, same as uploadOximeterCalibration() itself does internally
    if (max30102_connected) {
        uploadOximeterCalibration(current_hr, current_spo2, maxSensor.getIR(), maxSensor.getRed());
    }
    lastAccelUpload = now;
}

    // Periodic SpO2 update (more responsive than the 60s monitor cycle)
    if (max30102_connected && (now - lastSpo2Update >= SPO2_UPDATE_INTERVAL_MS)) {
        lastSpo2Update = now;
        long irValue = maxSensor.getIR();
        if (irValue > 25000) {
            updateSpO2();
        }
    }

    // Interval-based monitoring
    // Only trigger if we've passed the initial 30s warmup AND the interval is met
    // Interval-based monitoring
    if (now > 30000 && (now - lastMonitor >= currentMonitorInterval)) {
        lastMonitor = now;
        Serial.printf("\n[Monitor] Triggering analysis (interval: %lu ms)...\n", currentMonitorInterval);

        // 1. Audio Capture
        String label = captureAndClassifyAudio();

        // 2. Physiological Risk Calculation
        String overallRisk = calculatePhysioRisk(current_hr, current_spo2, label);
        if (overallRisk == "error") {
            // Keep monitoring reasonably often during sensor glitches.
            overallRisk = "medium";
        }

        // 7. Set Monitor Interval
        if (overallRisk == "critical") currentMonitorInterval = HIGH_RISK_MONITOR_MS;
        else if (overallRisk == "medium") currentMonitorInterval = MEDIUM_RISK_MONITOR_MS;
        else currentMonitorInterval = LOW_RISK_MONITOR_MS;

        // 8. Serial Monitoring (Cleaned up)
        Serial.printf("└─ [Physio]: %s | HR: %.1f | SpO2: %.1f%% | Audio: %s\n", overallRisk.c_str(), current_hr, current_spo2, label.c_str());

	delay(500); // let radio settle after HTTPS before ESP-NOW request
        // Request a fresh env snapshot so the status row updates together.
        (void)requestEnvSnapshot(1000);

        // 9. Uploads
        upload_device_status_upsert(current_hr, current_spo2, g_last_accel_magnitude, label, overallRisk);
        upload_sensor_row(current_hr, current_spo2, g_last_accel_magnitude, label, overallRisk);
        
        // 10. Data Reset
    }
}
